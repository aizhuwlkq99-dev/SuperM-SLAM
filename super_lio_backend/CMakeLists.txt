cmake_minimum_required(VERSION 3.16)
project(super_lio_backend LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(BACKEND_PREFIX "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/backend/install" ABSOLUTE)
list(PREPEND CMAKE_PREFIX_PATH "${BACKEND_PREFIX}")
set(CMAKE_INSTALL_RPATH "${BACKEND_PREFIX}/lib;\$ORIGIN/..;\$ORIGIN")
find_package(ament_cmake REQUIRED)
find_package(super_lio_mapping REQUIRED)
find_package(super_lio_interfaces REQUIRED)
find_package(GTSAM REQUIRED)
find_package(small_gicp REQUIRED)
find_package(TBB REQUIRED)
find_package(robin CONFIG REQUIRED)
find_package(kiss_matcher REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters kdtree)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(visualization_msgs REQUIRED)
add_library(superm_graph SHARED src/backend.cpp)
target_include_directories(superm_graph PUBLIC include ${PCL_INCLUDE_DIRS})
ament_target_dependencies(superm_graph super_lio_mapping)
target_link_libraries(superm_graph gtsam small_gicp::small_gicp kiss_matcher::kiss_matcher_core ${PCL_LIBRARIES})
add_executable(superm_optimize src/optimize_main.cpp)
target_link_libraries(superm_optimize superm_graph)
add_executable(superm_backend_node src/node.cpp)
target_link_libraries(superm_backend_node superm_graph)
ament_target_dependencies(superm_backend_node rclcpp nav_msgs sensor_msgs std_srvs std_msgs tf2_ros pcl_conversions visualization_msgs super_lio_interfaces)
install(TARGETS superm_graph LIBRARY DESTINATION lib)
add_executable(graph_check tests/graph_check.cpp)
target_link_libraries(graph_check superm_graph)
add_executable(superm_map_viewer src/map_viewer.cpp)
ament_target_dependencies(superm_map_viewer super_lio_mapping rclcpp sensor_msgs tf2_ros pcl_conversions)
install(TARGETS superm_optimize superm_backend_node graph_check superm_map_viewer DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
