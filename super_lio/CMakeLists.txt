cmake_minimum_required(VERSION 3.16)
project(super_lio)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(basic REQUIRED)
find_package(super_lio_mapping REQUIRED)
find_package(super_lio_interfaces REQUIRED)
find_package(m_detector REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED)
find_package(TBB REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/CloudPose.msg" "msg/CloudPose2.msg"
  DEPENDENCIES std_msgs sensor_msgs geometry_msgs)
file(GLOB LIO_SOURCES CONFIGURE_DEPENDS src/lio/*.cpp src/ros/*.cpp)
add_library(lio SHARED ${LIO_SOURCES})
target_compile_definitions(lio PUBLIC ROOT="${CMAKE_CURRENT_SOURCE_DIR}/")
target_include_directories(lio PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include> ${PCL_INCLUDE_DIRS})
ament_target_dependencies(lio super_lio_mapping super_lio_interfaces basic m_detector rclcpp sensor_msgs std_msgs geometry_msgs nav_msgs visualization_msgs pcl_conversions tf2_ros livox_ros_driver2)
rosidl_get_typesupport_target(typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(lio ${typesupport_target} Eigen3::Eigen ${PCL_LIBRARIES} TBB::tbb)
foreach(app super_lio_node relocation_node)
  add_executable(${app} src/apps/${app}.cpp)
  target_link_libraries(${app} lio)
  install(TARGETS ${app} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(TARGETS lio LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
get_filename_component(SUPERM_WORKSPACE_LOG_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../log" ABSOLUTE)
file(RELATIVE_PATH SUPERM_LOG_RELATIVE_DIR "${CMAKE_INSTALL_PREFIX}" "${SUPERM_WORKSPACE_LOG_DIR}")
configure_file(env-hooks/logging.sh.in "${CMAKE_CURRENT_BINARY_DIR}/logging.sh" @ONLY)
ament_environment_hooks("${CMAKE_CURRENT_BINARY_DIR}/logging.sh")
ament_export_dependencies(rosidl_default_runtime)
ament_package()
