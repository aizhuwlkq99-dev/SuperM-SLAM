cmake_minimum_required(VERSION 3.16)
project(m_detector)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(basic REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED)
find_package(OpenCV REQUIRED)
find_package(TBB REQUIRED)
find_package(OpenMP REQUIRED)
add_library(dyn_obj_filter SHARED src/DynObjFilter.cpp src/DynObjCluster.cpp)
target_include_directories(dyn_obj_filter PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include> ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(dyn_obj_filter basic rclcpp sensor_msgs std_msgs geometry_msgs nav_msgs visualization_msgs pcl_conversions)
target_link_libraries(dyn_obj_filter Eigen3::Eigen ${PCL_LIBRARIES} ${OpenCV_LIBS} TBB::tbb OpenMP::OpenMP_CXX)
install(TARGETS dyn_obj_filter EXPORT mDetectorTargets LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_export_targets(mDetectorTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(basic rclcpp sensor_msgs std_msgs geometry_msgs nav_msgs visualization_msgs pcl_conversions Eigen3 PCL OpenCV TBB OpenMP)
ament_package()
