cmake_minimum_required(VERSION 3.16)
project(basic)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
find_package(glog REQUIRED)
find_package(gflags REQUIRED)
add_library(basic SHARED src/Manifold.cpp src/logs.cpp)
target_include_directories(basic PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include> ${PCL_INCLUDE_DIRS})
target_link_libraries(basic Eigen3::Eigen ${PCL_LIBRARIES} glog::glog gflags)
ament_target_dependencies(basic rclcpp)
install(TARGETS basic EXPORT basicTargets LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(basicTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp Eigen3 PCL glog gflags)
ament_package()
